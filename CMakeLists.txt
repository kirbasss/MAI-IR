cmake_minimum_required(VERSION 3.20)
project(mai_ir LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(tokenizer_core STATIC cpp/tokenizer.cpp cpp/tokenizer_cli.cpp)
target_include_directories(tokenizer_core PUBLIC cpp)
if(MSVC)
    target_compile_options(tokenizer_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(tokenizer_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(tokenizer cpp/tokenizer_main.cpp)
target_link_libraries(tokenizer PRIVATE tokenizer_core)

include(CTest)
option(BUILD_DEMO "Build the native Lab 3 demonstration" ${BUILD_TESTING})
if(BUILD_TESTING OR BUILD_DEMO)
    find_package(nlohmann_json CONFIG REQUIRED)
endif()
if(BUILD_DEMO)
    add_library(tokenizer_demo_core STATIC cpp/tokenizer_demo.cpp)
    target_link_libraries(tokenizer_demo_core PUBLIC tokenizer_core nlohmann_json::nlohmann_json)
    add_executable(tokenizer_demo cpp/tokenizer_demo_main.cpp)
    target_link_libraries(tokenizer_demo PRIVATE tokenizer_demo_core)
    if(MSVC)
        target_compile_options(tokenizer_demo_core PRIVATE /W4 /utf-8)
        target_compile_options(tokenizer_demo PRIVATE /utf-8)
    else()
        target_compile_options(tokenizer_demo_core PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
if(BUILD_TESTING)
    # Dependencies are installed while building the Docker image, not at test time.
    find_package(GTest CONFIG REQUIRED)
    add_executable(tokenizer_tests tests/cpp/tokenizer_test.cpp tests/cpp/tokenizer_cli_test.cpp)
    target_link_libraries(tokenizer_tests PRIVATE tokenizer_core GTest::gtest_main nlohmann_json::nlohmann_json)
    target_compile_definitions(tokenizer_tests PRIVATE
        TOKENIZER_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    if(BUILD_DEMO)
        target_sources(tokenizer_tests PRIVATE tests/cpp/tokenizer_demo_test.cpp)
        target_link_libraries(tokenizer_tests PRIVATE tokenizer_demo_core)
    endif()
    if(MSVC)
        target_compile_options(tokenizer_tests PRIVATE /W4 /utf-8)
    else()
        target_compile_options(tokenizer_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    include(GoogleTest)
    gtest_discover_tests(tokenizer_tests DISCOVERY_TIMEOUT 30)
    add_test(NAME TokenizerProcess.Help COMMAND tokenizer --help)
    set_tests_properties(TokenizerProcess.Help PROPERTIES PASS_REGULAR_EXPRESSION "UTF-8")
    add_test(NAME TokenizerProcess.UnknownMode COMMAND tokenizer unknown)
    set_tests_properties(TokenizerProcess.UnknownMode PROPERTIES WILL_FAIL TRUE)
endif()
